			//PROCESS: this creates the HTML for the questionnaire for a single page
			function outputQuestionPage (questionPage) {
				var outputHTML	="";
				var questionObject	=SCREEN_ARRAY[questionPage];
				var title			=questionObject.title;
				var questionArray	=questionObject.questionArray;


				outputHTML	+="<h2>"+title+"</h2><form action='' name='formName'>";
				
				// loop through the question array, creating the interface for each part
				for (var questionIndex=0; questionIndex < questionArray.length; questionIndex++) {					
					//depending on which kind of input required...
					switch (questionArray[questionIndex].inputType) {
						case 'line':
							outputHTML	+=createInputLine(questionArray[questionIndex]);
							break;
						case 'header':
							outputHTML	+=createHeader(questionArray[questionIndex]);
							break;
						case 'date':
							outputHTML	+=createDate(questionArray[questionIndex]);						
							break;
						case 'checkbox':
							outputHTML	+=createCheckbox(questionArray[questionIndex]);						
							break;
						case 'radio':
							outputHTML	+=createRadio(questionArray[questionIndex]);						
							break;
						case 'linelist':
							outputHTML	+=createLineList(questionArray[questionIndex]);						
							break;
						case 'textbox':
							outputHTML	+=createTextbox(questionArray[questionIndex]);						
							break;
						case 'awardlist':
							outputHTML	+=createAwardList(questionArray[questionIndex]);						
							break;
						case 'dropdown':
							outputHTML	+=createDropdown(questionArray[questionIndex]);						
							break;
						default:
							outputHTML	+=createDefault(questionArray[questionIndex]);						
					}
				}
				
				//console.log('in outputQuestionPage!!!');
				
				var onClickAction	= 'displayQuestionPage('+ (questionPage+1) +');';
				if ((questionPage + 1) == TOTAL_PAGES) {
					onClickAction	= 'reviewResults('
									+ TOTAL_PAGES
									+ ');'
									+ onClickAction;
				}
				
				//console.log('TOTAL_PAGES:' + TOTAL_PAGES + 'questionPage +1:' + (questionPage+1) + 'onClickAction:' +onClickAction);
				
				outputHTML	+= '<div class="form-group">'
							+ '	<div class="control-label col-sm-2" ></div>'
							+ '	<div class="col-sm-4">'
							+ '		<button type="button" onclick="'
							+ onClickAction
							+ '">continue →</button>'
							+ '	</div>'
							+ '</div>'
							+ '</form>';
			
				outputPage(outputHTML, questionPage);
			}

			function reviewResults (resultsPage) {
				var responseObj	= {};

				for (var thisScreen in SCREEN_ARRAY) {
					var thisScreenArray	= SCREEN_ARRAY[thisScreen].questionArray;
					var questionClass 	= '';//'col-sm-2';
					var answerClass		= '';//'col-sm-10';
					
					// now, add the title for this section...
					outputHTML	+= "<dt class='" + questionClass + "'><h3>"
								+ SCREEN_ARRAY[thisScreen].title
								+ "</h3></dt>"
								+ "<dd class='" + answerClass + "'></dd>";
					
					// now loop through each part of this array...
					for (var thisQuestion in thisScreenArray) {
						var thisElement	= $('#' + thisScreenArray[thisQuestion].attribute);
						
						// if this is an existing element, then display it
						if (thisElement) {
							if (thisScreenArray[thisQuestion].attribute == 'null') {
								outputHTML	+= "<dt class='" + questionClass + "'><h4>"
											+ thisScreenArray[thisQuestion].attributeText
											+ "</h4></dt><dd class='" + answerClass + "'></dd>";
							} else {
								switch (thisScreenArray[thisQuestion].inputType) {
									case 'line':
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ thisElement.val()
													+ "</dd>";
													
										// add to the object of responses...
										responseObj[thisScreenArray[thisQuestion].attribute]	= thisElement.val();

										break;
									case 'textbox':
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ thisElement.val()
													+ "</dd>";
													
										// add to the object of responses...
										responseObj[thisScreenArray[thisQuestion].attribute]	= thisElement.val();

										break;
									case 'radio':
										var thisValue	= $('input[name="' + thisScreenArray[thisQuestion].attribute + '"]:checked').val();
								
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ thisValue
													+ "</dd>";
													
										// add to the object of responses...
																				
										if (thisValue == 'yes') {
											responseObj[thisScreenArray[thisQuestion].attribute]	= 1;
										} else if (thisValue == 'no') {
											responseObj[thisScreenArray[thisQuestion].attribute]	= 0;
										} else {
											responseObj[thisScreenArray[thisQuestion].attribute]	= thisValue;
										}

										break;
									case 'dropdown':
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ thisElement.val()
													+ "</dd>";
													
										// add to the object of responses...
										responseObj[thisScreenArray[thisQuestion].attribute]	= thisElement.val();

										break;
									case 'date':
										var monthNum	= $('#' + thisScreenArray[thisQuestion].attribute + '-month').val();
										monthNum		= parseInt(monthNum);
										var monthArray	='null, January, February, March, April, May, June, July, August, September, October, November, December'.split(', ');
										
										var sqlDate 	= $('#' + thisScreenArray[thisQuestion].attribute + '-year').val()
														+ '-'
														+ monthNum
														+ '-'
														+  $('#' + thisScreenArray[thisQuestion].attribute + '-date').val();
														
										var outputDate 	= monthArray[monthNum]
														+ ' '
														+  $('#' + thisScreenArray[thisQuestion].attribute + '-date').val()
														+ ", "
														+ $('#' + thisScreenArray[thisQuestion].attribute + '-year').val();
														
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ outputDate
													+ "</dd>";
													
										// add to the object of responses...
										responseObj[thisScreenArray[thisQuestion].attribute]	= sqlDate;

										break;
									case 'checkbox':
										var prompts			= thisScreenArray[thisQuestion].prompt.split('|');
										var answeredArray	= [];
										

										//then the options and values
										for (var thisPrompt in prompts) {

											var thisID	= '#' + thisScreenArray[thisQuestion].attribute + '_' + prompts[thisPrompt];
											thisID	=cleanID(thisID);

											var thisElement		= $(thisID);
											
											// add to object of responses...
											thisID		= thisID.replace('#', '');
												
											if (thisElement.is(':checked')) {													
												// add to the array of responses...
												answeredArray.push(prompts[thisPrompt]);
												
												responseObj[thisID]	= 1;
											} else {
												responseObj[thisID]	= 0;
											}
										}
										
										// add a combined answer to responseObj
										responseObj[thisScreenArray[thisQuestion].attribute]	= answeredArray.join(', ');
										
										// now to output the results
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ answeredArray.join('<br>')
													+ "</dd>";	

										break;
									case 'linelist':
										var prompt			= thisScreenArray[thisQuestion].prompt;
										var answeredArray	= [];
										
										//loop through each one of the line lists
										for (var thisPrompt =0; thisPrompt < prompt; thisPrompt++) {
											var thisID	= '#' + thisScreenArray[thisQuestion].attribute + '-' + thisPrompt;

											var thisElement		= $(thisID);
											var thisValue		= thisElement.val();
											
											//if there is a value in here then add to the list
											if (thisValue) {													
												// add to the array of responses...
												answeredArray.push(thisValue);
												
											}
										}
										
										//add to the response object
										var sqlList		= answeredArray.join(', ');
										responseObj[thisScreenArray[thisQuestion].attribute]	= sqlList;
										
										// now to output the results
										var htmlList	= answeredArray.join('<br />');
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ htmlList
													+ "</dd>";	

										break;
									case 'awardlist':
										var prompt			= thisScreenArray[thisQuestion].prompt;
										var answeredArray	= [];
										var valueTotal	= 0;
										
										//loop through each one of the line lists
										for (var thisPrompt =0; thisPrompt < prompt; thisPrompt++) {
										
											// grab the names and values...
											var valueID	= '#value-' + thisScreenArray[thisQuestion].attribute + '-' + thisPrompt;
											var nameID	= '#name-' + thisScreenArray[thisQuestion].attribute + '-' + thisPrompt;

											var thisValueElement	= $(valueID);
											var thisNameElement		= $(nameID);
											var thisValueValue		= thisValueElement.val();
											var thisNameValue		= thisNameElement.val();

											if (thisValueValue) {													
												valueTotal	+= parseInt(thisValueValue);
												answeredArray.push('$' + thisValueValue + ' - ' + thisNameValue);
											}
										}
										
										//add to the response object
										var sqlList		= answeredArray.join(', ');
										responseObj[thisScreenArray[thisQuestion].attribute]					= sqlList;
										responseObj[thisScreenArray[thisQuestion].attribute + '_totalValue']	= valueTotal;
										
										// now to output the results
										var htmlList	= answeredArray.join('<br />');
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>"
													+ htmlList
													+ "</dd>";	

										break;
									default:
										outputHTML	+= "<dt class='" + questionClass + "'>"
													+ thisScreenArray[thisQuestion].attributeText
													+ "</dt>"
													+ "<dd class='" + answerClass + "'>DEFFF!"
													+ thisScreenArray[thisQuestion].attribute
													+ 'val:'
													+ thisElement.val()
													+ "</dd>";

										break;
								}
							}
						
						} else { // if this is not an existing element..

			function createCheckbox(questionObject) {//PROCESS: this creates the HTML for a collection of checkboxes
				var outputHTML	='<div class="form-group">'
										+'	<label class="control-label col-sm-2" for="' +cleanID(questionObject.attribute)+ '">'+ htmlize(questionObject.attributeText) +'</label>'
										+'	<div class="col-sm-10">';
										
				var optionList	=questionObject.prompt.split('|');//get the list of checkbox items from .prompt - seperated by |
				
				for (var i=0; i< optionList.length; i++) {//loop through each checkbox item
					var thisID	= cleanID(questionObject.attribute)
								+ "_"
								+ cleanID(optionList[i]);	
					thisID	=cleanID(thisID);
					
					outputHTML	+='<div class="checkbox">'
								+"	<label>"
								+"	<input type='checkbox' id='" 
								+ thisID
								+ "' value='"
								+ htmlize(optionList[i])
								+ "'>"
								+ htmlize(optionList[i])
								+"	</label>"
								+'</div>';
				}
				
				outputHTML	+='	</div>'
							+'</div>';
							
				return outputHTML;
			}
			
			function cleanID (thisString) {
				thisString		= thisString.replace(/ /g, '_');
				thisString		= thisString.replace(/\'/, '');
				thisString		= thisString.replace(/\(/g, '');
				thisString		= thisString.replace(/\)/g, '');
				thisString		= thisString.replace(/\-/g, '_');
				thisString		= thisString.replace(/\,/g, '');
				thisString		= thisString.replace(/\./g, '');
				thisString		= thisString.replace(/\//g, '_or_');

				return thisString;
			}
